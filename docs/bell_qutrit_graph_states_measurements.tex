\section{Zakres}

Ten dokument zbiera tylko te fragmenty z pracy \texttt{pdfs/scalableameoperators.pdf}, kt\'ore s\k{a}
potrzebne do opisania qutrytowych operator\'ow pomiarowych, qutrytowych stan\'ow grafowych,
konstrukcji nier\'owno\'sci Bella oraz przypadk\'ow: dw\'och qutryt\'ow, AME(4,3) i
3-qutrytowego GHZ. Nie jest to streszczenie ca\l{}ego artyku\l{}u.

\section{Scenariusz Bella}

Rozwa\.zamy $N$ odleg\l{}ych obserwator\'ow $A_i$, kt\'orzy wsp\'olnie dziel\k{a} stan
$\rho \in \mathcal{H}_1\otimes\cdots\otimes\mathcal{H}_N$. Obserwator $A_i$ wybiera
ustawienie pomiaru $x_i$ i otrzymuje wynik $a_i \in \{0,\ldots,d-1\}$. Korelacje s\k{a}
zadane przez
\begin{equation}
    p(\vec{a}\mid\vec{x})
    =
    \operatorname{Tr}\!\left[
        \rho\left(M^1_{a_1\mid x_1}\otimes\cdots\otimes M^N_{a_N\mid x_N}\right)
    \right].
\end{equation}
Zachowanie jest lokalne, je\.zeli mo\.zna je zapisa\'c jako
\begin{equation}
    p(\vec{a}\mid\vec{x})
    =
    \sum_{\lambda}\mu(\lambda)
    p_1(a_1\mid x_1,\lambda)\cdots p_N(a_N\mid x_N,\lambda).
\end{equation}
Og\'olna nier\'owno\'s\'c Bella ma posta\'c
\begin{equation}
    I=\sum_{\vec{a},\vec{x}}\alpha_{\vec{a},\vec{x}}p(\vec{a}\mid\vec{x})
    \leq \beta_L,
\end{equation}
gdzie $\beta_L$ jest ograniczeniem klasycznym. Ograniczenie kwantowe definiujemy jako
\begin{equation}
    \beta_Q=\sup_{\vec{p}\in Q_{N,m,d}} I.
\end{equation}

Artyku\l{} u\.zywa tak\.ze zespolonych warto\'sci oczekiwanych, czyli dyskretnej transformaty
Fouriera rozk\l{}ad\'ow prawdopodobie\'nstwa:
\begin{equation}
    \left\langle A^1_{n_1\mid x_1}\cdots A^N_{n_N\mid x_N}\right\rangle
    =
    \sum_{\vec{a}}\omega^{\vec{a}\cdot\vec{n}}p(\vec{a}\mid\vec{x}),
    \qquad
    \omega=\exp(2\pi i/d).
\end{equation}
Dla qutryt\'ow $d=3$, wi\k{e}c $\omega=\exp(2\pi i/3)$.

\section{Qutrytowe operatory $X$, $Z$ i $\omega$}

Dla qutrytu baz\k{a} obliczeniow\k{a} jest $\{\ket{0},\ket{1},\ket{2}\}$. Operatory przesuni\k{e}cia
i fazy s\k{a} zdefiniowane przez
\begin{align}
    X\ket{j} &= \ket{j+1 \bmod 3},\\
    Z\ket{j} &= \omega^j\ket{j},\\
    \omega &= \exp(2\pi i/3).
\end{align}
Spe\l{}niaj\k{a} one $\omega^3=1$, $1+\omega+\omega^2=0$ oraz
\begin{equation}
    XZ=\omega^{-1}ZX.
\end{equation}

\section{Stany grafowe i stabilizatory}

Niech $G=(V,E,R,d)$ b\k{e}dzie grafem o lokalnym wymiarze pierwszym $d$, gdzie
$R=(r_{ij})$ opisuje liczno\'s\'c kraw\k{e}dzi modulo $d$. Operator stabilizuj\k{a}cy wierzcho\l{}ek
$i$ ma posta\'c
\begin{equation}
    G_i = X_i\prod_{j\in \mathcal{N}_i} Z_j^{r_{ij}},
\end{equation}
gdzie $\mathcal{N}_i$ jest s\k{a}siedztwem wierzcho\l{}ka $i$. Stan grafowy $\ket{G}$ jest
jedynym wsp\'olnym stanem w\l{}asnym z warto\'sci\k{a} w\l{}asn\k{a} $1$:
\begin{equation}
    G_i\ket{G}=\ket{G}, \qquad i=1,\ldots,N.
\end{equation}

Dla dw\'och qutryt\'ow po\l{}\k{a}czonych jedn\k{a} kraw\k{e}dzi\k{a} stabilizatory maj\k{a} posta\'c
\begin{equation}
    G_1=X\otimes Z,\qquad G_2=Z\otimes X.
\end{equation}
Po lokalnej transformacie Fouriera drugi r\'ownowa\.zny zestaw dla stanu maksymalnie
splotanego $\ket{\psi^+_3}$ to
\begin{equation}
    G'_1=X\otimes X,\qquad G'_2=Z\otimes Z^\dagger.
\end{equation}

\section{Regu\l{}a zamiany operator\'ow dla $d=3$}

Dla dowolnego pierwszego $d$ artyku\l{} wprowadza obserwable
\begin{equation}
    O_k=XZ^k,\qquad k=0,\ldots,d-1.
\end{equation}
W konstrukcji nier\'owno\'sci Bella operatory stabilizatorowe zawieraj\k{a}ce
$(XZ^k)^n$ s\k{a} zast\k{e}powane przez kombinacje obserwabli pomiarowych. Dla qutryt\'ow:
\begin{equation}
    (XZ^k)^n \longrightarrow \widetilde{A}^{(n)}_k,
    \qquad k=0,1,2,\quad n=1,2.
\end{equation}
Wygodna posta\'c jawna, zgodna z r\'ownaniem zamiany z PDF-a, to
\begin{equation}
    \widetilde{A}^{(n)}_k
    =
    \frac{\omega^{-n k(k+1)/2}}{\sqrt{3}\lambda_n}
    \sum_{t=0}^{2}\omega^{-ntk}A_t^n,
    \qquad k=0,1,2,\quad n=1,2,
    \label{eq:qutrit-replacement}
\end{equation}
gdzie czynnik $1/2$ jest rozumiany modulo $3$. PDF podaje
\begin{equation}
    \lambda_1=-i\omega^{2/3}=\omega^{1/12}=\exp(\pi i/18),
    \qquad
    \lambda_2=\lambda_1^*=\exp(-\pi i/18).
\end{equation}

Dla $n=1$ PDF zapisuje to jawnie jako
\begin{align}
    X &\longrightarrow \widetilde{A}_0
    =\frac{1}{\sqrt{3}\lambda_1}(A_0+A_1+A_2),\\
    XZ &\longrightarrow \widetilde{A}_1
    =\frac{1}{\sqrt{3}\lambda_1\omega}
    (A_0+\omega^{-1}A_1+\omega^{-2}A_2),\\
    XZ^2 &\longrightarrow \widetilde{A}_2
    =\frac{1}{\sqrt{3}\lambda_1}
    (A_0+\omega^{-2}A_1+\omega^{-1}A_2).
\end{align}
Dla $n=2$ obowi\k{a}zuje
\begin{equation}
    (XZ^k)^2=(XZ^k)^\dagger,
    \qquad
    \widetilde{A}^{(2)}_k=(\widetilde{A}^{(1)}_k)^\dagger,
    \qquad k=0,1,2.
\end{equation}

\section{Dwa qutryty}

Dla stanu maksymalnie spl\k{a}tanego dw\'och qutryt\'ow artyku\l{} bierze stabilizatory
\begin{equation}
    G_k^n=
    \left[G'_1(G'_2)^k\right]^n
    =
    (XZ^k)^n\otimes(XZ^{-k})^n,
    \qquad k=0,1,2,\quad n=1,2.
\end{equation}
Zamiana na obserwable Bella jest
\begin{equation}
    (XZ^k)^n\longrightarrow \widetilde{A}^{(n)}_k,
    \qquad
    (XZ^{-k})^n\longrightarrow B_k^n.
\end{equation}
St\k{a}d wyra\.zenie Bella z PDF-a dla $d=3$ ma posta\'c
\begin{align}
    I_{\max}^{(3)}
    &:=
    \sum_{n=1}^{2}\sum_{k=0}^{2}
    \left\langle \widetilde{A}^{(n)}_k B_k^n \right\rangle \\
    &=
    \frac{1}{\sqrt{3}}
    \sum_{n=1}^{2}\lambda_n^*
    \sum_{x,y=0}^{2}
    \omega^{-nxy}
    \left\langle A_x^n B_y^n \right\rangle
    \leq \beta_L^{\max}.
\end{align}
PDF potwierdza w tabeli dla $d=3$:
\begin{equation}
    \beta_Q=3(3-1)=6,
    \qquad
    \beta_L=6\cos(\pi/9)\approx 5.63816.
\end{equation}

\section{AME(4,3)}

Dla grafu AME(4,3) stabilizatory z PDF-a s\k{a} nast\k{e}puj\k{a}ce:
\begin{align}
    G_1 &= X_1 Z_2 Z_4,\\
    G_2 &= Z_1 X_2 Z_3,\\
    G_3 &= Z_2 X_3 Z_4^2,\\
    G_4 &= Z_1 Z_3^2 X_4.
\end{align}
Do konstrukcji nier\'owno\'sci PDF u\.zywa zestawu
\begin{align}
    G_1 &= X\otimes Z\otimes \id\otimes Z,\\
    G_1G_2 &= XZ\otimes ZX\otimes Z\otimes Z,\\
    G_1G_2^2 &= XZ^2\otimes ZX^2\otimes Z^2\otimes Z,\\
    G_3 &= \id\otimes Z\otimes X\otimes Z^2,\\
    G_1G_4 &= XZ\otimes Z\otimes Z^2\otimes ZX.
\end{align}
Odpowiadaj\k{a}ce podstawienia pomiarowe s\k{a}
\begin{equation}
    Z\to B_0,\quad ZX\to B_1,\quad ZX^2\to B_2,
    \qquad
    Z^k\to C_0^k,\quad X\to C_1,
    \qquad
    Z\to D_0,\quad ZX\to D_1.
\end{equation}

Bell expression z PDF-a jest
\begin{align}
    I_{\mathrm{AME}}
    &:=
    \frac{1}{\sqrt{3}\lambda_1}
    \Big[
        \left\langle (A_0+A_1+A_2)B_0D_0 \right\rangle
        +
        \left\langle (A_0+\omega^2A_1+\omega A_2)B_2C_0^2D_0 \right\rangle
    \Big] \notag\\
    &\quad+
    \frac{1}{2\sqrt{3}\lambda_1\omega}
    \Big[
        \left\langle (A_0+\omega A_1+\omega^2A_2)B_1C_0D_0 \right\rangle
        +
        \left\langle (A_0+\omega A_1+\omega^2A_2)B_0C_0^2D_1 \right\rangle
    \Big] \notag\\
    &\quad+
    \left\langle B_0C_1D_0\right\rangle+\cc
    \leq \beta_L^{\mathrm{AME}}.
\end{align}
PDF podaje ograniczenie klasyczne oraz dowodzi ograniczenia kwantowego:
\begin{equation}
    \beta_L^{\mathrm{AME}}=7.63816,
    \qquad
    \beta_Q^{\mathrm{AME}}=8.
\end{equation}

\section{Og\'olna konstrukcja dla stan\'ow grafowych}

Dla og\'olnego grafu wybiera si\k{e} dwa po\l{}\k{a}czone wierzcho\l{}ki oznaczone jako $1$ i $2$.
Niech $\mathcal{N}_1$ oznacza s\k{a}siedztwo wierzcho\l{}ka $1$, a $N_1=|\mathcal{N}_1|$.
Zestaw stabilizator\'ow u\.zywany do konstrukcji ma posta\'c
\begin{align}
    G_{1,k}^n &:= (G_1G_2^k)^n,
    && k=0,\ldots,d-1,\\
    G_{2,k}^n &:= (G_1G_k)^n,
    && k=3,\ldots,N_1+1,\\
    G_{3,k}^n &:= G_k^n,
    && k=N_1+2,\ldots,N,
\end{align}
gdzie $n=1,\ldots,d-1$. Po zastosowaniu regu\l{}y zamiany dla pierwszego wierzcho\l{}ka oraz
bezpo\'srednich podstawie\'n obserwabli na pozosta\l{}ych wierzcho\l{}kach PDF otrzymuje rodzin\k{e}
wyra\.ze\'n Bella
\begin{equation}
    I_G =
    \sum_{n=1}^{d-1}
    \left[
        \left\langle \widetilde{G}^{(n)}_{1,0}\right\rangle
        +
        \sum_{k=1}^{d-1}c_{1,k}
        \left\langle \widetilde{G}^{(n)}_{1,k}\right\rangle
        +
        \sum_{k=3}^{N_1+1}c_{2,k}
        \left\langle \widetilde{G}^{(n)}_{2,k}\right\rangle
        +
        \sum_{k=N_1+2}^{N}
        \left\langle \widetilde{G}^{(n)}_{3,k}\right\rangle
    \right],
\end{equation}
przy dodatnich wsp\'olczynnikach spe\l{}niaj\k{a}cych warunek normalizacji z PDF-a:
\begin{equation}
    c_{1,k}
    +
    \sum_{\substack{j=3\\ r_{1,j}=kr_{1,2}}}^{N_1+1} c_{2,j}
    =
    1,
    \qquad k=1,\ldots,d-1.
\end{equation}
Dla tej rodziny artyku\l{} dowodzi
\begin{equation}
    \beta_Q^G=(d-1)(N-N_1+d-1).
\end{equation}

\section{3-qutrytowy GHZ jako stan grafowy}

PDF opisuje $N$-qutrytowy GHZ jako stan grafowy dla grafu-gwiazdy. Dla $N=3$ i $d=3$
wybieram wierzcho\l{}ek centralny jako wierzcho\l{}ek $1$; dla trzech wierzcho\l{}k\'ow ta gwiazda
jest jednocze\'snie grafem-lini\k{a} z centrum w \'srodku. Stabilizatory s\k{a}
\begin{align}
    G_1 &= X_1Z_2Z_3,\\
    G_2 &= Z_1X_2,\\
    G_3 &= Z_1X_3.
\end{align}
Mamy wtedy $N=3$, $d=3$ i $N_1=2$, wi\k{e}c og\'olna formu\l{}a daje
\begin{align}
    \beta_Q^G
    &=
    (d-1)(N-N_1+d-1)\\
    &=
    (3-1)(3-2+3-1)\\
    &=6.
\end{align}
Klasyczne ograniczenie dla tego konkretnego 3-qutrytowego przypadku GHZ nie jest w PDF-ie
podane jawnie. PDF stwierdza og\'olnie, \.ze wyznaczenie $\beta_L$ dla rodziny $I_G$ jest trudne,
a warto\'sci klasyczne podaje numerycznie tylko dla AME(4,3) oraz dla najprostszych
dwukuditowych przypadk\'ow.

\section{Kodowanie qutrytu w dw\'och qubitach przez izometri\k{e} $E$}

W implementacji pojedynczy qutryt jest kodowany w dw\'och qubitach przez izometri\k{e}
\begin{equation}
    E:\mathbb{C}^3\longrightarrow \mathbb{C}^4,
    \qquad
    E^\dagger E=\id_3.
\end{equation}
Kolumny macierzy $E$ s\k{a} zakodowanymi stanami bazowymi qutrytu. Projektor na
podprzestrze\'n kodow\k{a} jednego bloku to
\begin{equation}
    P_E=EE^\dagger,
    \qquad
    P_E^2=P_E=P_E^\dagger.
\end{equation}
Bloki qubitowe s\k{a} przyjmowane jako
\begin{equation}
    q_0=[0,1],\qquad q_1=[2,3],\qquad q_2=[4,5],\qquad q_3=[6,7].
\end{equation}
Nie nale\.zy zak\l{}ada\'c, \.ze $E=E_ZW$ ani \.ze podprzestrze\'n kodowa jest zawsze
$\operatorname{span}\{\ket{00},\ket{01},\ket{10}\}$. Og\'olne $E$ mo\.ze mie\'c support na
$\ket{11}$, a wi\k{e}c interpretacja bitstringu jako $0,1,2$ oraz leakage jako samego wyniku
$11$ jest poprawna tylko dla szczeg\'olnego kodowania obliczeniowego.

\section{Transformacja operator\'ow i projektor\'ow do kodowania $E$}

Dla operatora qutrytowego $O\in\mathbb{C}^{3\times 3}$ operator dzia\l{}aj\k{a}cy na
fizycznym bloku dw\'och qubit\'ow jest
\begin{equation}
    O_{\mathrm{phys}}=E O E^\dagger.
\end{equation}
Jest to operator wsparty na podprzestrzeni kodowej. Je\.zeli potrzebna jest pe\l{}na macierz
$4\times4$, kt\'ora dzia\l{}a jak $O$ na kodzie i jak identyczno\'s\'c poza kodem, u\.zywamy
\begin{equation}
    O_{\mathrm{full}}=E O E^\dagger+I_\perp,
    \qquad
    I_\perp=\id_4-EE^\dagger.
\end{equation}
Je\.zeli $O$ jest unitarny na $\mathbb{C}^3$, to $O_{\mathrm{full}}$ jest unitarny na
$\mathbb{C}^4$. Ta posta\'c jest w\l{}a\'sciwa dla obserwabli typu $X$, $Z$, $XZ^k$,
$A_x$, $B_y$, $C_z$, $D_z$ wbudowywanych do obwodu qubitowego.

Dla projektor\'ow pomiarowych $P_a=\ket{v_a}\bra{v_a}$:
\begin{equation}
    P_{a,\mathrm{phys}}=E P_a E^\dagger,
    \qquad
    \sum_{a=0}^{2}P_{a,\mathrm{phys}}=P_E.
\end{equation}
Projektory wynik\'ow qutrytowych nie sumuj\k{a} si\k{e} do $\id_4$, tylko do projektora kodu.
Pozosta\l{}a przestrze\'n $I_\perp$ jest leakage.

Dla $N$ qutryt\'ow projektor kodu wynosi
\begin{equation}
    P_{\mathrm{code}}=P_E^{\otimes N}.
\end{equation}
Prawdopodobie\'nstwo leakage dla stanu fizycznego $\ket{\psi}$:
\begin{equation}
    p_{\mathrm{leak}}
    =
    1-\bra{\psi}P_E^{\otimes N}\ket{\psi}.
\end{equation}

W trybie Estimator nale\.zy raportowa\'c $p_{\mathrm{leak}}$ obok warto\'sci Bella. Je\.zeli
$p_{\mathrm{leak}}$ nie jest bliski zera, wynik nie jest bezpo\'srednim wynikiem qutrytowym.
W trybie Sampler s\k{a} dwie poprawne strategie:
\begin{itemize}
    \item traktowa\'c leakage jako osobny wynik b\l{}\k{e}du i raportowa\'c jego cz\k{e}sto\'s\'c;
    \item postselekcjonowa\'c na podprzestrze\'n kodow\k{a}, ale wtedy jawnie poda\'c
    $p_{\mathrm{leak}}$ i liczy\'c warto\'sci oczekiwane z rozk\l{}adu warunkowego.
\end{itemize}
Dla og\'olnego $E$ Sampler powinien mierzy\'c projektory
$\{P_{0,\mathrm{phys}},P_{1,\mathrm{phys}},P_{2,\mathrm{phys}},I_\perp\}$ albo najpierw
zastosowa\'c dekoder unitarny $U_E^\dagger$, kt\'ory przenosi kod na trzy wybrane stany
obliczeniowe i stan ortogonalny na etykiet\k{e} leakage.

\section{Estimator}

Operatory z PDF-a s\k{a} cz\k{e}sto unitarne i zespolone, ale nie s\k{a} hermitowskie.
Estimator powinien dostawa\'c obserwable hermitowskie. Dla dowolnego operatora $O$
stosujemy rozk\l{}ad
\begin{align}
    \operatorname{Re}(O)&=\frac{O+O^\dagger}{2},\\
    \operatorname{Im}(O)&=\frac{O-O^\dagger}{2i}.
\end{align}
Wtedy
\begin{equation}
    \langle O\rangle
    =
    \langle \operatorname{Re}(O)\rangle
    +
    i\langle \operatorname{Im}(O)\rangle.
\end{equation}
Alternatywnie mo\.zna od razu budowa\'c hermitowski operator Bella jako
\begin{equation}
    B_{\mathrm{Herm}}=\sum_s \alpha_s O_s+\alpha_s^*O_s^\dagger.
\end{equation}
Ta druga forma odpowiada zapisowi ``$+\cc$'' w PDF-ie. W obu podej\'sciach trzeba
sprawdza\'c numerycznie
\begin{equation}
    B_{\mathrm{Herm}}^\dagger=B_{\mathrm{Herm}}.
\end{equation}

\section{Sampler}

Sampler nie mierzy bezpo\'srednio niehermitowskiej warto\'sci oczekiwanej. Dla danego
ustawienia pomiarowego nale\.zy:
\begin{enumerate}
    \item zdiagonalizowa\'c qutrytowy operator pomiarowy $A_x$ albo jego wersj\k{e}
    $A_{x,\mathrm{full}}=E A_x E^\dagger+I_\perp$;
    \item zbudowa\'c zmian\k{e} bazy, kt\'ora mierzy w bazie w\l{}asnej tego operatora;
    \item dla ka\.zdego bloku dw\'och qubit\'ow zmapowa\'c wynik do $a_i\in\{0,1,2\}$
    albo do etykiety leakage;
    \item z rozk\l{}adu wynik\'ow policzy\'c zespolon\k{a} warto\'s\'c oczekiwan\k{a}
    \begin{equation}
        \left\langle \prod_i A_i^{n_i}\right\rangle
        =
        \sum_{\vec{a}\in\{0,1,2\}^N}
        \omega^{\vec{a}\cdot\vec{n}}p(\vec{a}\mid\vec{x}).
    \end{equation}
\end{enumerate}
Je\.zeli $E$ jest og\'olne, proste mapowanie bitstring\'ow
$00\mapsto0$, $01\mapsto1$, $10\mapsto2$, $11\mapsto$ leakage nie jest wystarczaj\k{a}ce.
Najpierw trzeba przej\'s\'c do bazy, w kt\'orej podprzestrze\'n kodowa jest rozdzielona od
$I_\perp$.

\section{Audyt statyczny notebook\'ow}

Audyt wykonano statycznie przez odczyt kom\'orek JSON notebook\'ow. Nie uruchamiano
pe\l{}nych notebook\'ow. Uruchomiono tylko ma\l{}y fragment sprawdzaj\k{a}cy warto\'sci
$\lambda_1,\lambda_2$ dla $d=3$; wynik zgadza si\k{e} z
$\lambda_1=\exp(\pi i/18)$ i $\lambda_2=\exp(-\pi i/18)$.

\subsection{\texttt{notebooks/2qutrit\_RyGates.ipynb}}

Notebook pr\'obuje liczy\'c Bell expression dla dw\'och qutryt\'ow, startuj\k{a}c od obwodu
\texttt{create\_ame\_circuit(2, 3)}. Odpowiada to dwukuditowemu przypadkowi z PDF-a:
\begin{equation}
    I_{\max}^{(3)}=\sum_{n=1}^{2}\sum_{k=0}^{2}
    \left\langle\widetilde{A}^{(n)}_kB_k^n\right\rangle,
    \qquad \beta_Q=6.
\end{equation}
Zapisane outputy notebooka zawieraj\k{a} warto\'s\'c oko\l{}o
$6.000000000000009$, czyli zgodn\k{a} z boundem kwantowym.

Funkcja \texttt{A\_t(t,n)} buduje obserwable $A_t^n$ z odwr\'oconej transformacji PDF-a,
a nie sam operator $\widetilde{A}^{(n)}_k$. To jest poprawne dla realizacji kwantowej,
bo $\widetilde{A}^{(n)}_k$ pojawia si\k{e} dopiero jako kombinacja tych obserwabli w
wyra\.zeniu Bella. Funkcja \texttt{B\_k(k,n)} u\.zywa $(ZX^k)^n$; trzeba pilnowa\'c zgodno\'sci
konwencji z PDF-em, gdzie dla drugiej strony wyst\k{e}puje $(XZ^{-k})^n$.

Notebook traktuje $n=1$ i $n=2$ oddzielnie przez listy \texttt{op} i \texttt{op2}. Czynnik
c.c. jest wi\k{e}c realizowany jako cz\k{e}\'s\'c $n=2$, a nie przez jawne dodanie
$O^\dagger$ w jednym operatorze. Do Estimatora nie s\k{a} podawane surowe operatory
niehermitowskie: \texttt{create\_isa\_ops} rozbija ka\.zdy tensor na
\texttt{0.5*(AB+AB.adjoint())} oraz \texttt{-0.5j*(AB-AB.adjoint())}.

Mapowanie qubit\'ow jest po\'srednio zgodne z blokami $[0,1]$ i $[2,3]$, ale kod nie
formalizuje izometrii $E$ ani leakage. U\.zywa gotowych macierzy \texttt{Xgate3},
\texttt{Zgate3} jako operator\'ow $4\times4$, czyli jest zwi\k{a}zany z konkretnym kodowaniem.
Dla og\'olnego $E$ trzeba zast\k{a}pi\'c to konstrukcj\k{a} $EOE^\dagger+I_\perp$.
Rotacje \texttt{Ry01/Ry02/Ry12} s\k{a} pobocznym skanem optymalizacyjnym po lokalnym
bloku qutrytu, a nie elementem konstrukcji PDF.

\subsection{\texttt{notebooks/ghz\_ry\_chsh\_recovered.ipynb}}

Notebook buduje 3-qutrytowy stan grafowy przez
\texttt{create\_ame\_circuit(3, 3, graph\_type='line')} i liczy wariant $I_G$ dla grafu
GHZ/line. Zapisane wyniki statevector/Estimator pokazuj\k{a} maksimum oko\l{}o $6$,
zgodne z formu\l{}\k{a} PDF-a dla $d=3$, $N=3$, $N_1=2$.

Operator \texttt{A\_t(t,n)} ma tak\k{a} sam\k{a} rol\k{e} jak w notebooku dwqutrytowym:
buduje realizacyjne obserwable $A_t^n$. \texttt{B\_k(k,n)} i \texttt{C\_k(k,n)} implementuj\k{a}
operatory potrzebne dla wybranego grafu; w szczeg\'olno\'sci \texttt{C\_k(1,n)} u\.zywa
$(ZX)^n$, co odpowiada operatorowi typu $Z^{r_{1,k}}X$ w og\'olnej konstrukcji.

Notebook jawnie uwzgl\k{e}dnia $n=1$ i $n=2$ i buduje cz\k{e}\'s\'c c.c. przez drug\k{a} list\k{e}
operator\'ow. Dla Estimatora ponownie stosuje rozk\l{}ad na cz\k{e}\'s\'c rzeczywist\k{a}
i urojon\k{a}, wi\k{e}c nie przekazuje bezpo\'srednio operator\'ow niehermitowskich.
Trzeba jednak pilnowa\'c kolejno\'sci tensor\'ow: w kodzie wyst\k{e}puje \texttt{B \^{} A \^{} C},
co musi by\'c zgodne z konwencj\k{a} Qiskit dla indeksowania qubit\'ow i layoutu. Dokument
przyjmuje bloki $[0,1]$, $[2,3]$, $[4,5]$.

Tak jak poprzednio, notebook nie modeluje og\'olnego $E$ ani leakage. Importuje Sampler,
ale g\l{}\'owna rachunkowa \'scie\.zka jest Estimator/Statevector. Rotacje
\texttt{Ry01/Ry02/Ry12} s\k{a} pobocznym skanem lokalnych zmian bazy, nie cz\k{e}\'sci\k{a}
nier\'owno\'sci z PDF-a.

\subsection{\texttt{notebooks/ame43\_random\_chsh.ipynb}}

Notebook pr\'obuje liczy\'c przypadek AME(4,3): buduje graf AME, tworzy obw\'od przez
\texttt{create\_ame\_circuit(dim=3, graph=game43)}, liczy klasyczny bound przez brute force
oraz warto\'s\'c kwantow\k{a} przez Statevector i Estimator. Zapisane outputy zawieraj\k{a}
$\beta_L\approx7.6381557247$ oraz wynik Statevector/Estimator oko\l{}o $8$, zgodnie z PDF-em.

Struktura operator\'ow odpowiada wyra\.zeniu AME z PDF-a: listy w \texttt{create\_isa\_ops}
zawieraj\k{a} 13 tensor\'ow dla $n=1$ i 13 tensor\'ow dla $n=2$, zgodnych z pi\k{e}cioma
cz\l{}onami AME po rozwini\k{e}ciu sum po $A_0,A_1,A_2$. Notebook poprawnie rozbija
operatory na cz\k{e}\'s\'c rzeczywist\k{a} i urojon\k{a} przed \texttt{SparsePauliOp.from\_operator}.

Najwa\.zniejsze ograniczenie: notebook u\.zywa konkretnych bramek \texttt{Xgate3},
\texttt{Zgate3} jako macierzy $4\times4$, a nie jawnej izometrii $E$. W zwi\k{a}zku z tym
nie sprawdza $E^\dagger E$, projektora $P_E$ ani $p_{\mathrm{leak}}$. Dla og\'olnego
kodowania trzeba zast\k{a}pi\'c operatory przez $EOE^\dagger+I_\perp$ i mierzy\'c leakage.

W outputach z sesji sprz\k{e}towych pojawiaj\k{a} si\k{e} \texttt{NaN} oraz pojedynczy wynik

sk\l{}adnik\'ow, b\l{}\k{e}dy agregacji, \texttt{NaN} albo brak kontroli leakage/noise. Rotacje
\texttt{Ry01/Ry02/Ry12} s\k{a} skanem lokalnej optymalizacji i nie nale\.z\k{a} do konstrukcji PDF.

def make_XZ_qutrit():
    omega = exp(2j*pi/3)
    X = zeros((3, 3), complex)
    Z = diag([1, omega, omega**2])
    for j in range(3):
        X[(j + 1) % 3, j] = 1
    return X, Z, omega

def make_A_tilde_qutrit_d3(A, k, n):
    # A = [A_0, A_1, A_2], each A_t is a qutrit observable.
    omega = exp(2j*pi/3)
    lambda_n = exp((1 if n == 1 else -1) * 1j*pi/18)
    pref = omega ** (-n * k * (k + 1) / 2) / (sqrt(3) * lambda_n)
    return pref * sum((omega ** (-n * t * k)) * matrix_power(A[t], n)
                      for t in range(3))

def embed_operator_E(E, O, full=True):
    # E has shape (4, 3) and E.conj().T @ E == I_3.
    O_phys = E @ O @ E.conj().T
    if not full:
        return O_phys
    I_perp = eye(4) - E @ E.conj().T
    return O_phys + I_perp

def embed_projector_E(E, P):
    return E @ P @ E.conj().T

def split_nonhermitian(O):
    O_re = 0.5 * (O + O.conj().T)
    O_im = (O - O.conj().T) / (2j)
    return O_re, O_im

def build_bell_operator_two_qutrit(E):
    X, Z, omega = make_XZ_qutrit()
    A = build_measurement_observables_from_pdf(E)
    terms = []
    for n in [1, 2]:
        for k in [0, 1, 2]:
            Atilde = make_A_tilde_qutrit_d3(A, k, n)
            B = matrix_power(X @ matrix_power(Z, -k % 3), n)
            terms.append(kron(embed_operator_E(E, Atilde, full=True),
                              embed_operator_E(E, B, full=True)))
    return sum(terms)

def build_bell_operator_ghz_graph(E, graph):
    terms = build_general_graph_terms_d3(graph, E)
    B = sum(weight * term for weight, term in terms)
    return B + B.conj().T

def build_bell_operator_ame43(E):
    terms_n1 = build_ame43_pdf_terms(E, n=1)
    terms_n2 = build_ame43_pdf_terms(E, n=2)
    return sum(terms_n1) + sum(terms_n2)

def bell_value_estimator(estimator, circuit, B):
    Bre, Bim = split_nonhermitian(B)
    ev_re = estimator.run([(circuit, Bre)]).result()[0].data.evs
    ev_im = estimator.run([(circuit, Bim)]).result()[0].data.evs
    return ev_re + 1j * ev_im

def bell_value_sampler(counts, settings, powers):
    omega = exp(2j*pi/3)
    total = 0
    good = 0
    leak = 0
    for bitstring, c in counts.items():
        outcome = decode_blocks_or_leak(bitstring, settings)
        if outcome.is_leak:
            leak += c
            continue
        phase = omega ** dot(outcome.qutrit_values, powers)
        total += c * phase
        good += c
    return total / good, leak / (good + leak)

def leakage_probability(state_or_estimator, E, N):
    P_E = E @ E.conj().T
    P_code = kron_power(P_E, N)
    return 1 - expectation(P_code, state_or_estimator)

\section{Checklist walidacji implementacji}

\begin{itemize}
    \item Sprawdzi\'c numerycznie $E^\dagger E\approx \id_3$.
    \item Sprawdzi\'c, \.ze $P_E=EE^\dagger$ jest projektorem:
    $P_E^2\approx P_E$ i $P_E^\dagger\approx P_E$.
    \item Raportowa\'c $p_{\mathrm{leak}}\approx 0$ dla symulacji idealnych.
    \item Sprawdzi\'c, \.ze osadzone obserwable $EOE^\dagger+I_\perp$ s\k{a} unitarne.
    \item Sprawdzi\'c, \.ze $\operatorname{Re}(O)$ i $\operatorname{Im}(O)$ s\k{a}
    hermitowskie.
    \item Sprawdzi\'c, \.ze operator Bella po dodaniu c.c./dagger jest hermitowski.
    \item Dla szczeg\'olnego kodowania $E=E_Z$ por\'owna\'c wyniki z PDF-em i obecnymi
    notebookami.
    \item Dwa qutryty: w symulacji idealnej otrzyma\'c $\beta_Q=6$.
    \item AME(4,3): w symulacji idealnej otrzyma\'c $\beta_Q=8$.
    \item GHZ: w symulacji idealnej otrzyma\'c $\beta_Q$ z formu\l{}y PDF-a
    $(d-1)(N-N_1+d-1)$, czyli $6$ dla wyboru $d=3$, $N=3$, $N_1=2$.
\end{itemize}
